export const DeconStepTracker = () => {
  const [completedSteps, setCompletedSteps] = useState([])
  const [activeTimer, setActiveTimer] = useState(null)
  const [timeLeft, setTimeLeft] = useState(0)
  
  const steps = [
    { id: 1, title: 'Remove Clothing', icon: <Shirt className="w-5 h-5" />, desc: 'Remove outer clothing first, cut off if needed', result: 'Removes ~90% of contamination', timer: null },
    { id: 2, title: 'Bag & Seal', icon: <Shield className="w-5 h-5" />, desc: 'Place contaminated items in plastic bag, seal tightly', result: 'Prevents further spread', timer: null },
    { id: 3, title: 'Rinse Body', icon: <Droplets className="w-5 h-5" />, desc: 'Rinse with lukewarm water, head to toe', result: 'Removes loose particles', timer: 120 },
    { id: 4, title: 'Wash with Soap', icon: <Sparkles className="w-5 h-5" />, desc: 'Wash entire body with soap, focus on hair and nails', result: 'Deep cleaning', timer: 180 },
    { id: 5, title: 'Dry & Dress', icon: <User className="w-5 h-5" />, desc: 'Dry with clean towel, put on clean clothes', result: 'Decontamination complete', timer: null },
  ]
  
  useEffect(() => {
    if (activeTimer !== null && timeLeft > 0) {
      const interval = setInterval(() => {
        setTimeLeft(t => t - 1)
      }, 1000)
      return () => clearInterval(interval)
    } else if (timeLeft === 0 && activeTimer !== null) {
      setCompletedSteps([...completedSteps, activeTimer])
      setActiveTimer(null)
    }
  }, [activeTimer, timeLeft])
  
  const startTimer = (stepId, seconds) => {
    setActiveTimer(stepId)
    setTimeLeft(seconds)
  }
  
  const toggleStep = (stepId) => {
    if (completedSteps.includes(stepId)) {
      setCompletedSteps(completedSteps.filter(s => s !== stepId))
    } else {
      setCompletedSteps([...completedSteps, stepId])
    }
  }
  
  const formatTime = (seconds) => {
    const mins = Math.floor(seconds / 60)
    const secs = seconds % 60
    return `${mins}:${secs.toString().padStart(2, '0')}`
  }
  
  const progress = (completedSteps.length / steps.length) * 100
  
  return (
    <div className="my-6 p-4 bg-gray-800/50 rounded-lg">
      <div className="flex items-center justify-between mb-4">
        <h4 className="text-lg font-bold text-green-300">Decontamination Progress</h4>
        <span className="text-sm text-gray-400">{completedSteps.length}/{steps.length} complete</span>
      </div>
      
      {/* Progress bar */}
      <div className="h-3 bg-gray-700 rounded-full mb-6 overflow-hidden">
        <div 
          className="h-full bg-gradient-to-r from-green-500 to-emerald-400 transition-all duration-500"
          style={{ width: `${progress}%` }}
        />
      </div>
      
      <div className="space-y-3">
        {steps.map((step, index) => {
          const isComplete = completedSteps.includes(step.id)
          const isActive = activeTimer === step.id
          const isPrevComplete = index === 0 || completedSteps.includes(steps[index - 1].id)
          
          return (
            <div 
              key={step.id}
              className={`p-4 rounded-lg transition-all ${
                isComplete 
                  ? 'bg-green-900/30 border border-green-500' 
                  : isActive 
                    ? 'bg-blue-900/30 border border-blue-500'
                    : 'bg-gray-900/50 border border-gray-700'
              }`}
            >
              <div className="flex items-center gap-4">
                <button
                  onClick={() => step.timer && !isComplete ? startTimer(step.id, step.timer) : toggleStep(step.id)}
                  disabled={!isPrevComplete && !isComplete}
                  className={`w-10 h-10 rounded-full flex items-center justify-center transition-all ${
                    isComplete 
                      ? 'bg-green-600' 
                      : isActive 
                        ? 'bg-blue-600 animate-pulse'
                        : isPrevComplete 
                          ? 'bg-gray-600 hover:bg-gray-500' 
                          : 'bg-gray-700 opacity-50'
                  }`}
                >
                  {isComplete ? (
                    <CheckCircle className="w-6 h-6" />
                  ) : isActive ? (
                    <Clock className="w-6 h-6" />
                  ) : (
                    step.icon
                  )}
                </button>
                
                <div className="flex-1">
                  <div className="flex items-center gap-2">
                    <h5 className={`font-bold ${isComplete ? 'text-green-400' : ''}`}>
                      {step.title}
                    </h5>
                    {step.timer && !isComplete && (
                      <span className="text-xs bg-blue-600 px-2 py-0.5 rounded">
                        {step.timer}s timer
                      </span>
                    )}
                  </div>
                  <p className="text-sm text-gray-400">{step.desc}</p>
                  {isComplete && (
                    <p className="text-sm text-green-400 mt-1">✓ {step.result}</p>
                  )}
                </div>
                
                {isActive && (
                  <div className="text-right">
                    <div className="text-3xl font-mono font-bold text-blue-400">
                      {formatTime(timeLeft)}
                    </div>
                    <p className="text-xs text-gray-400">remaining</p>
                  </div>
                )}
              </div>
            </div>
          )
        })}
      </div>
      
      {completedSteps.length === steps.length && (
        <div className="mt-4 p-4 bg-green-900/30 border border-green-500 rounded-lg text-center">
          <CheckCircle className="w-8 h-8 mx-auto text-green-400 mb-2" />
          <p className="text-green-400 font-bold text-lg">Decontamination Complete!</p>
          <p className="text-green-200 text-sm">Continue monitoring for symptoms as advised</p>
        </div>
      )}
    </div>
  )
}

export const EyeFlushTimer = () => {
  const [isRunning, setIsRunning] = useState(false)
  const [timeElapsed, setTimeElapsed] = useState(0)
  
  const targetTime = 900 // 15 minutes in seconds
  
  useEffect(() => {
    if (isRunning && timeElapsed < targetTime) {
      const interval = setInterval(() => {
        setTimeElapsed(t => t + 1)
      }, 1000)
      return () => clearInterval(interval)
    }
  }, [isRunning, timeElapsed])
  
  const formatTime = (seconds) => {
    const mins = Math.floor(seconds / 60)
    const secs = seconds % 60
    return `${mins}:${secs.toString().padStart(2, '0')}`
  }
  
  const progress = (timeElapsed / targetTime) * 100
  const isComplete = timeElapsed >= targetTime
  
  return (
    <div className="my-6 p-4 bg-gray-800/50 rounded-lg">
      <div className="flex items-center gap-2 text-cyan-400 font-bold mb-4">
        <Eye className="w-6 h-6" />
        <h4 className="text-lg">Eye Flush Timer (15 min minimum)</h4>
      </div>
      
      <div className="text-center">
        <div className={`text-6xl font-mono font-bold mb-4 ${isComplete ? 'text-green-400' : 'text-cyan-400'}`}>
          {formatTime(timeElapsed)}
        </div>
        
        <div className="h-4 bg-gray-700 rounded-full mb-4 overflow-hidden">
          <div 
            className={`h-full transition-all duration-500 ${isComplete ? 'bg-green-500' : 'bg-cyan-500'}`}
            style={{ width: `${Math.min(progress, 100)}%` }}
          />
        </div>
        
        <p className="text-gray-400 mb-4">
          {isComplete 
            ? '✓ Minimum flush time reached' 
            : `${formatTime(targetTime - timeElapsed)} remaining (minimum)`}
        </p>
        
        <div className="flex gap-2 justify-center">
          <button
            onClick={() => setIsRunning(!isRunning)}
            className={`px-6 py-3 rounded-lg font-bold transition-all ${
              isRunning 
                ? 'bg-yellow-600 hover:bg-yellow-500' 
                : 'bg-cyan-600 hover:bg-cyan-500'
            }`}
          >
            {isRunning ? 'Pause' : timeElapsed > 0 ? 'Resume' : 'Start Timer'}
          </button>
          <button
            onClick={() => { setIsRunning(false); setTimeElapsed(0) }}
            className="px-6 py-3 bg-gray-600 rounded-lg hover:bg-gray-500"
          >
            Reset
          </button>
        </div>
        
        <div className="mt-4 text-sm text-gray-400 text-left">
          <p><strong>Instructions:</strong></p>
          <ul className="list-disc list-inside mt-1 space-y-1">
            <li>Flush from inner corner to outer corner</li>
            <li>Use clean, cool water</li>
            <li>Hold eyelids open during flushing</li>
            <li>Do NOT rub eyes</li>
            <li>Seek medical care after flushing</li>
          </ul>
        </div>
      </div>
    </div>
  )
}

```
DECONTAMINATION STEPS

1. REMOVE clothing → Removes ~90% of contamination

3. WASH with soap → Hair with shampoo, no conditioner

EYES: Flush 15-20 minutes, inner to outer corner
WOUNDS: Cover before washing, avoid contaminating
```
