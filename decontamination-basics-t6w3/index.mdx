export const DeconStepTracker = () => {
  const [completedSteps, setCompletedSteps] = useState([])
  const [activeTimer, setActiveTimer] = useState(null)
  const [timeLeft, setTimeLeft] = useState(0)
  
  const steps = [
    { id: 1, title: 'Remove Clothing', icon: <Shirt className="w-5 h-5" />, desc: 'Outer layer first. Cut off anything that goes over your head.', result: 'Removes about 80-90% of contamination', timer: null },
    { id: 2, title: 'Bag & Seal', icon: <Shield className="w-5 h-5" />, desc: 'Double-bag clothing, jewelry, and contact lenses. Not in the regular trash.', result: 'Keeps it from spreading', timer: null },
    { id: 3, title: 'Wash with Soap', icon: <Droplets className="w-5 h-5" />, desc: 'Lukewarm water and mild soap, head to toe. Do not scrub. Keep cuts covered.', result: 'Removes most of what is left', timer: 180 },
    { id: 4, title: 'Face, Eyes, Hair', icon: <Sparkles className="w-5 h-5" />, desc: 'Blow your nose; wipe eyelids and ears. Radiation: shampoo, no conditioner. Eyes burning? Use the eye flush timer below.', result: 'Face and hair cleaned', timer: null },
    { id: 5, title: 'Dry & Dress', icon: <User className="w-5 h-5" />, desc: 'Dry with a clean towel, put on clean clothes, bag the towel', result: 'Decontamination complete', timer: null },
  ]
  
  useEffect(() => {
    if (activeTimer !== null && timeLeft > 0) {
      const interval = setInterval(() => {
        setTimeLeft(t => t - 1)
      }, 1000)
      return () => clearInterval(interval)
    } else if (timeLeft === 0 && activeTimer !== null) {
      setCompletedSteps([...completedSteps, activeTimer])
      setActiveTimer(null)
    }
  }, [activeTimer, timeLeft])
  
  const startTimer = (stepId, seconds) => {
    setActiveTimer(stepId)
    setTimeLeft(seconds)
  }
  
  const toggleStep = (stepId) => {
    if (completedSteps.includes(stepId)) {
      setCompletedSteps(completedSteps.filter(s => s !== stepId))
    } else {
      setCompletedSteps([...completedSteps, stepId])
    }
  }
  
  const formatTime = (seconds) => {
    const mins = Math.floor(seconds / 60)
    const secs = seconds % 60
    return `${mins}:${secs.toString().padStart(2, '0')}`
  }
  
  const progress = (completedSteps.length / steps.length) * 100
  
  return (
    <div className="my-6 p-4 bg-gray-800/50 rounded-lg">
      <div className="flex items-center justify-between mb-4">
        <h4 className="text-lg font-bold text-green-300">Decontamination Progress</h4>
        <span className="text-sm text-gray-400">{completedSteps.length}/{steps.length} complete</span>
      </div>
      
      {/* Progress bar */}
      <div className="h-3 bg-gray-700 rounded-full mb-6 overflow-hidden">
        <div 
          className="h-full bg-gradient-to-r from-green-500 to-emerald-400 transition-all duration-500"
          style={{ width: `${progress}%` }}
        />
      </div>
      
      <div className="space-y-3">
        {steps.map((step, index) => {
          const isComplete = completedSteps.includes(step.id)
          const isActive = activeTimer === step.id
          const isPrevComplete = index === 0 || completedSteps.includes(steps[index - 1].id)
          
          return (
            <div 
              key={step.id}
              className={`p-4 rounded-lg transition-all ${
                isComplete 
                  ? 'bg-green-900/30 border border-green-500' 
                  : isActive 
                    ? 'bg-blue-900/30 border border-blue-500'
                    : 'bg-gray-900/50 border border-gray-700'
              }`}
            >
              <div className="flex items-center gap-4">
                <button
                  onClick={() => step.timer && !isComplete ? startTimer(step.id, step.timer) : toggleStep(step.id)}
                  disabled={!isPrevComplete && !isComplete}
                  className={`w-10 h-10 rounded-full flex items-center justify-center transition-all ${
                    isComplete 
                      ? 'bg-green-600' 
                      : isActive 
                        ? 'bg-blue-600 animate-pulse'
                        : isPrevComplete 
                          ? 'bg-gray-600 hover:bg-gray-500' 
                          : 'bg-gray-700 opacity-50'
                  }`}
                >
                  {isComplete ? (
                    <CheckCircle className="w-6 h-6" />
                  ) : isActive ? (
                    <Clock className="w-6 h-6" />
                  ) : (
                    step.icon
                  )}
                </button>
                
                <div className="flex-1">
                  <div className="flex items-center gap-2">
                    <h5 className={`font-bold ${isComplete ? 'text-green-400' : ''}`}>
                      {step.title}
                    </h5>
                    {step.timer && !isComplete && (
                      <span className="text-xs bg-blue-600 px-2 py-0.5 rounded">
                        {step.timer}s timer
                      </span>
                    )}
                  </div>
                  <p className="text-sm text-gray-400">{step.desc}</p>
                  {isComplete && (
                    <p className="text-sm text-green-400 mt-1">✓ {step.result}</p>
                  )}
                </div>
                
                {isActive && (
                  <div className="text-right">
                    <div className="text-3xl font-mono font-bold text-blue-400">
                      {formatTime(timeLeft)}
                    </div>
                    <p className="text-xs text-gray-400">remaining</p>
                  </div>
                )}
              </div>
            </div>
          )
        })}
      </div>
      
      {completedSteps.length === steps.length && (
        <div className="mt-4 p-4 bg-green-900/30 border border-green-500 rounded-lg text-center">
          <CheckCircle className="w-8 h-8 mx-auto text-green-400 mb-2" />
          <p className="text-green-400 font-bold text-lg">Decontamination Complete!</p>
          <p className="text-green-200 text-sm">Call Poison Control (1-800-222-1222) or 911 if you have symptoms</p>
        </div>
      )}
    </div>
  )
}

export const EyeFlushTimer = () => {
  const [isRunning, setIsRunning] = useState(false)
  const [timeElapsed, setTimeElapsed] = useState(0)
  
  const targetTime = 900 // 15 minutes in seconds
  
  useEffect(() => {
    if (isRunning && timeElapsed < targetTime) {
      const interval = setInterval(() => {
        setTimeElapsed(t => t + 1)
      }, 1000)
      return () => clearInterval(interval)
    }
  }, [isRunning, timeElapsed])
  
  const formatTime = (seconds) => {
    const mins = Math.floor(seconds / 60)
    const secs = seconds % 60
    return `${mins}:${secs.toString().padStart(2, '0')}`
  }
  
  const progress = (timeElapsed / targetTime) * 100
  const isComplete = timeElapsed >= targetTime
  
  return (
    <div className="my-6 p-4 bg-gray-800/50 rounded-lg">
      <div className="flex items-center gap-2 text-cyan-400 font-bold mb-4">
        <Eye className="w-6 h-6" />
        <h4 className="text-lg">Eye Flush Timer (10-15 minutes)</h4>
      </div>
      
      <div className="text-center">
        <div className={`text-6xl font-mono font-bold mb-4 ${isComplete ? 'text-green-400' : 'text-cyan-400'}`}>
          {formatTime(timeElapsed)}
        </div>
        
        <div className="h-4 bg-gray-700 rounded-full mb-4 overflow-hidden">
          <div 
            className={`h-full transition-all duration-500 ${isComplete ? 'bg-green-500' : 'bg-cyan-500'}`}
            style={{ width: `${Math.min(progress, 100)}%` }}
          />
        </div>
        
        <p className="text-gray-400 mb-4">
          {isComplete 
            ? '✓ 15 minutes reached. Keep flushing if it still burns, and get medical care.'
            : `${formatTime(targetTime - timeElapsed)} remaining`}
        </p>
        
        <div className="flex gap-2 justify-center">
          <button
            onClick={() => setIsRunning(!isRunning)}
            className={`px-6 py-3 rounded-lg font-bold transition-all ${
              isRunning 
                ? 'bg-yellow-600 hover:bg-yellow-500' 
                : 'bg-cyan-600 hover:bg-cyan-500'
            }`}
          >
            {isRunning ? 'Pause' : timeElapsed > 0 ? 'Resume' : 'Start Timer'}
          </button>
          <button
            onClick={() => { setIsRunning(false); setTimeElapsed(0) }}
            className="px-6 py-3 bg-gray-600 rounded-lg hover:bg-gray-500"
          >
            Reset
          </button>
        </div>
        
        <div className="mt-4 text-sm text-gray-400 text-left">
          <p><strong>Instructions:</strong></p>
          <ul className="list-disc list-inside mt-1 space-y-1">
            <li>Take out contact lenses first</li>
            <li>Hold eyelids open; let water run from the inner corner (by the nose) outward</li>
            <li>Use clean, lukewarm water</li>
            <li>Do NOT rub your eyes or use eye drops</li>
            <li>Get medical care after flushing</li>
          </ul>
        </div>
      </div>
    </div>
  )
}

# Decontamination Basics

<CriticalAlert>
**If a chemical, radioactive dust, or a biological agent got on you, get it off fast.** Taking off your clothes removes most of it; washing with soap and water removes most of the rest. Call **911** for trouble breathing, seizures, collapse, or burns. **Poison Control (US): 1-800-222-1222.**
</CriticalAlert>

## Do This Now

1. **Get away from the source** and into fresh air or a safe building (for radiation, get inside).
2. **Take off your clothes**, starting with the outer layer, ideally within 10 minutes for a chemical. This removes about 80-90% of the contamination. **Cut off** anything you'd have to pull over your head; if you can't, hold your breath and close your eyes while you pull it off.
3. **Bag it.** Put clothes, jewelry, and contact lenses in a plastic bag, close it, and put that in a second bag. Keep it away from people and pets, and out of the regular trash.
4. **Wash** with lukewarm water and mild soap, head to toe, for at least 2-3 minutes. **Don't scrub**, scald, or scratch your skin. Keep water out of your eyes, nose, and mouth. Keep cuts covered.
5. **No shower?** Blot (don't rub) your skin, face, hands, and hair with a moist wipe, wet cloth, or damp paper towel.
6. **Eyes burning or vision blurry?** Flush with clean water for 10-15 minutes. No eye drops.
7. **Put on clean clothes** that were stored away from the contamination.
8. **Get help:** call 911 or Poison Control if you have symptoms, can't get the chemical off, or swallowed it.

**If a chemical was swallowed:** don't make yourself throw up and don't drink anything else. Call Poison Control or 911.

**Tap water is fine for washing** even if officials say not to drink it.

## Special Situations

- **Wounds:** keep cuts and scrapes covered while you wash, and get medical care for wounds that may be contaminated.
- **Glasses:** after a chemical, soak them in household bleach, then rinse and dry before wearing.
- **Children:** put on a mask and gloves if you have them, then help them through every step. Cut clothing off rather than pulling it over their heads. Keep them warm afterward.
- **Pets:** wear gloves. For radioactive dust, gently brush the coat, then wash with shampoo or soap and water. Wash your own hands and face afterward. Ask a veterinarian about chemical exposure.
- **Cold weather:** dry off and dress quickly; washing can chill you fast.

```
DECONTAMINATION STEPS
```
