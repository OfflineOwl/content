export const DecisionTool = () => {
  const [answers, setAnswers] = useState({})
  
  const questions = [
    { id: 'sick', question: 'Are you currently showing symptoms of illness?', yes: 'isolation', no: 'continue' },
    { id: 'confirmed', question: 'Have you been diagnosed with a contagious disease?', yes: 'isolation', no: 'continue' },
    { id: 'exposed', question: 'Have you been in close contact with someone who is sick?', yes: 'quarantine', no: 'continue' },
    { id: 'travel', question: 'Have you traveled from an area with an active outbreak?', yes: 'quarantine', no: 'neither' },
  ]
  
  const getResult = () => {
    for (const q of questions) {
      const answer = answers[q.id]
      if (answer === 'yes' && (q.yes === 'isolation' || q.yes === 'quarantine')) {
        return q.yes
      }
    }
    
    const allAnswered = questions.every(q => answers[q.id] !== undefined)
    if (allAnswered) return 'neither'
    return null
  }
  
  const result = getResult()
  const currentQuestion = questions.find(q => answers[q.id] === undefined && !result)
  
  const resetQuiz = () => setAnswers({})
  
  return (
    <div className="my-6 p-4 bg-gray-800/50 rounded-lg">
      <h4 className="text-lg font-bold text-blue-300 mb-4">Do I Need Isolation or Quarantine?</h4>
      
      {!result && currentQuestion && (
        <div className="space-y-4">
          <div className="p-4 bg-gray-900/50 rounded-lg">
            <p className="text-lg mb-4">{currentQuestion.question}</p>
            <div className="flex gap-3">
              <button
                onClick={() => setAnswers({ ...answers, [currentQuestion.id]: 'yes' })}
                className="flex-1 py-3 bg-blue-600 hover:bg-blue-500 rounded-lg font-bold transition-all"
              >
                Yes
              </button>
              <button
                onClick={() => setAnswers({ ...answers, [currentQuestion.id]: 'no' })}
                className="flex-1 py-3 bg-gray-600 hover:bg-gray-500 rounded-lg font-bold transition-all"
              >
                No
              </button>
            </div>
          </div>
          
          {/* Progress dots */}
          <div className="flex justify-center gap-2">
            {questions.map((q, i) => (
              <div
                key={q.id}
                className={`w-3 h-3 rounded-full ${
                  answers[q.id] !== undefined 
                    ? 'bg-blue-500' 
                    : q.id === currentQuestion.id 
                      ? 'bg-blue-400 animate-pulse' 
                      : 'bg-gray-600'
                }`}
              />
            ))}
          </div>
        </div>
      )}
      
      {result && (
        <div className={`p-6 rounded-lg text-center ${
          result === 'isolation' 
            ? 'bg-red-900/30 border-2 border-red-500' 
            : result === 'quarantine'
              ? 'bg-yellow-900/30 border-2 border-yellow-500'
              : 'bg-green-900/30 border-2 border-green-500'
        }`}>
          {result === 'isolation' && (
            <>
              <BedDouble className="w-12 h-12 mx-auto mb-3 text-red-400" />
              <h5 className="text-2xl font-bold text-red-400 mb-2">ISOLATION Recommended</h5>
              <p className="text-red-200">You are sick or have been diagnosed. Separate yourself from others until you've recovered.</p>
            </>
          )}
          {result === 'quarantine' && (
            <>
              <Home className="w-12 h-12 mx-auto mb-3 text-yellow-400" />
              <h5 className="text-2xl font-bold text-yellow-400 mb-2">QUARANTINE Recommended</h5>
              <p className="text-yellow-200">You've been exposed and may become sick. Stay home and monitor for symptoms.</p>
            </>
          )}
          {result === 'neither' && (
            <>
              <CheckCircle className="w-12 h-12 mx-auto mb-3 text-green-400" />
              <h5 className="text-2xl font-bold text-green-400 mb-2">No Special Measures Needed</h5>
              <p className="text-green-200">Based on your answers, you don't currently need isolation or quarantine.</p>
            </>
          )}
          
          <button
            onClick={resetQuiz}
            className="mt-4 px-4 py-2 bg-gray-600 hover:bg-gray-500 rounded-lg text-sm"
          >
            Start Over
          </button>
        </div>
      )}
    </div>
  )
}

export const SymptomTracker = () => {
  const [days, setDays] = useState([])
  const [currentDay, setCurrentDay] = useState(1)
  
  const symptoms = [
    { id: 'fever', label: 'Fever (100.4°F+)', icon: <Thermometer className="w-4 h-4" /> },
    { id: 'cough', label: 'Cough', icon: <Activity className="w-4 h-4" /> },
    { id: 'breathing', label: 'Difficulty breathing', icon: <Heart className="w-4 h-4" /> },
    { id: 'fatigue', label: 'Fatigue', icon: <BedDouble className="w-4 h-4" /> },
    { id: 'bodyaches', label: 'Body aches', icon: <User className="w-4 h-4" /> },
    { id: 'other', label: 'Other symptoms', icon: <AlertCircle className="w-4 h-4" /> },
  ]
  
  const [todaySymptoms, setTodaySymptoms] = useState({})
  const [temperature, setTemperature] = useState('')
  
  const toggleSymptom = (symptomId) => {
    setTodaySymptoms(prev => ({
      ...prev,
      [symptomId]: !prev[symptomId]
    }))
  }
  
  const saveDay = () => {
    const hasSymptoms = Object.values(todaySymptoms).some(v => v)
    const entry = {
      day: currentDay,
      symptoms: { ...todaySymptoms },
      temperature: temperature || 'Not recorded',
      hasSymptoms
    }
    setDays([...days, entry])
    setCurrentDay(currentDay + 1)
    setTodaySymptoms({})
    setTemperature('')
  }
  
  const hasDangerSigns = Object.values(todaySymptoms).filter(v => v).length >= 3 || todaySymptoms.breathing
  
  return (
    <div className="my-6 p-4 bg-gray-800/50 rounded-lg">
      <h4 className="text-lg font-bold text-blue-300 mb-4 flex items-center gap-2">
        <Activity className="w-5 h-5" />
        Daily Symptom Tracker (Day {currentDay})
      </h4>
      
      {/* Previous days summary */}
      {days.length > 0 && (
        <div className="mb-4 flex gap-1 flex-wrap">
          {days.map((d, i) => (
            <div 
              key={i}
              className={`w-8 h-8 rounded flex items-center justify-center text-xs font-bold ${
                d.hasSymptoms 
                  ? 'bg-red-600' 
                  : 'bg-green-600'
              }`}
              title={`Day ${d.day}: ${d.hasSymptoms ? 'Had symptoms' : 'No symptoms'}`}
            >
              {d.day}
            </div>
          ))}
          <div className="w-8 h-8 rounded bg-blue-600 flex items-center justify-center text-xs font-bold animate-pulse">
            {currentDay}
          </div>
        </div>
      )}
      
      {/* Temperature input */}
      <div className="mb-4">
        <label className="text-sm text-gray-400 mb-1 block">Temperature (°F)</label>
        <input
          type="number"
          step="0.1"
          value={temperature}
          onChange={(e) => setTemperature(e.target.value)}
          placeholder="98.6"
          className="w-32 px-3 py-2 bg-gray-900 border border-gray-600 rounded text-white"
        />
        {parseFloat(temperature) >= 100.4 && (
          <span className="ml-2 text-red-400 text-sm">⚠️ Fever detected</span>
        )}
      </div>
      
      {/* Symptoms checklist */}
      <div className="grid grid-cols-2 md:grid-cols-3 gap-2 mb-4">
        {symptoms.map(s => (
          <button
            key={s.id}
            onClick={() => toggleSymptom(s.id)}
            className={`p-3 rounded-lg text-left transition-all flex items-center gap-2 ${
              todaySymptoms[s.id] 
                ? 'bg-red-600 ring-2 ring-red-400' 
                : 'bg-gray-700 hover:bg-gray-600'
            }`}
          >
            {s.icon}
            <span className="text-sm">{s.label}</span>
          </button>
        ))}
      </div>
      
      {/* Warning if danger signs */}
      {hasDangerSigns && (
        <div className="mb-4 p-3 bg-red-900/50 border border-red-500 rounded-lg">
          <AlertTriangle className="w-5 h-5 inline mr-2 text-red-400" />
          <span className="text-red-200 font-bold">Seek medical attention if symptoms are severe or worsening</span>
        </div>
      )}
      
      <button
        onClick={saveDay}
        className="w-full py-3 bg-blue-600 hover:bg-blue-500 rounded-lg font-bold transition-all"
      >
        Log Day {currentDay} & Continue
      </button>
      
      {days.length >= 5 && !days.slice(-3).some(d => d.hasSymptoms) && (
        <div className="mt-4 p-4 bg-green-900/30 border border-green-500 rounded-lg text-center">
          <CheckCircle className="w-8 h-8 mx-auto text-green-400 mb-2" />
          <p className="text-green-400 font-bold">Looking Good!</p>
          <p className="text-green-200 text-sm">No symptoms for several days. Consult health guidance for when to end quarantine.</p>
        </div>
      )}
    </div>
  )
}

## 📋 Quick Reference

```
ISOLATION = SICK
├── Confirmed illness
├── Stay separated until recovered  
├── Caregiver uses PPE
└── Dedicated room/bathroom

QUARANTINE = EXPOSED  
├── Was exposed, may not be sick yet
├── Monitor for symptoms daily
├── Stay home during incubation period
└── May still develop illness

BOTH REQUIRE:
├── Staying home
├── Separation from others
├── Monitoring health
└── Following official guidance
```
