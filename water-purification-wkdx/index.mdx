export const AltitudeCalculator = () => {
  const [altitude, setAltitude] = useState(0)
  
  const getBoilTime = (alt) => {
    if (alt < 6500) return 1
    return 3
  }
  
  const getBoilTemp = (alt) => {
    // Approximate: loses about 1°F per 500 feet
    const tempDrop = Math.floor(alt / 500)
    return Math.max(185, 212 - tempDrop)
  }
  
  const altitudes = [
    { value: 0, label: 'Sea Level' },
    { value: 5000, label: '5,000 ft' },
    { value: 6500, label: '6,500 ft' },
    { value: 10000, label: '10,000 ft' },
    { value: 15000, label: '15,000 ft' },
  ]
  
  return (
    <div className="my-6 p-4 bg-gray-800/50 rounded-lg">
      <div className="flex items-center gap-2 text-amber-400 font-bold mb-4">
        <Mountain className="w-6 h-6" />
        <h4 className="text-lg">Altitude Boiling Calculator</h4>
      </div>
      
      <div className="flex flex-wrap gap-2 mb-4">
        {altitudes.map(a => (
          <button
            key={a.value}
            onClick={() => setAltitude(a.value)}
            className={`px-4 py-2 rounded-lg transition-all ${
              altitude === a.value 
                ? 'bg-amber-600 text-white' 
                : 'bg-gray-700 hover:bg-gray-600'
            }`}
          >
            {a.label}
          </button>
        ))}
      </div>
      
      <div className="grid md:grid-cols-3 gap-4">
        <div className="p-4 bg-gray-900/50 rounded-lg text-center">
          <p className="text-gray-400 text-sm">Boiling Point</p>
          <p className="text-3xl font-mono font-bold text-amber-400">{getBoilTemp(altitude)}°F</p>
          <p className="text-gray-500 text-sm">{Math.round((getBoilTemp(altitude) - 32) * 5/9)}°C</p>
        </div>
        <div className="p-4 bg-gray-900/50 rounded-lg text-center">
          <p className="text-gray-400 text-sm">Required Boil Time</p>
          <p className="text-3xl font-mono font-bold text-orange-400">{getBoilTime(altitude)} min</p>
          <p className="text-gray-500 text-sm">rolling boil</p>
        </div>
        <div className="p-4 bg-gray-900/50 rounded-lg text-center">
          <p className="text-gray-400 text-sm">Altitude</p>
          <p className="text-3xl font-mono font-bold text-blue-400">{altitude.toLocaleString()}</p>
          <p className="text-gray-500 text-sm">feet</p>
        </div>
      </div>
      
      {altitude >= 6500 && (
        <div className="mt-4 p-3 bg-amber-900/30 border border-amber-500 rounded-lg text-amber-200 text-sm">
          <AlertTriangle className="w-4 h-4 inline mr-2" />
          At high altitude, water boils at a lower temperature. Boil for 3 minutes to ensure safety.
        </div>
      )}
    </div>
  )
}

      </div>
      
      <div className="flex justify-between mt-4">

          className="px-4 py-2 bg-gray-600 rounded hover:bg-gray-500 disabled:opacity-50"

        </span>
        <button 
          onClick={() => setCurrentStep(Math.min(current.steps.length - 1, currentStep + 1))}
          className="px-4 py-2 bg-green-600 rounded hover:bg-green-500 disabled:opacity-50"
          disabled={currentStep === current.steps.length - 1}
        >
          Next
        </button>
      </div>
    </div>
  )
}

|-----------|-------------|

## 🔄 Combining Methods (Maximum Safety)

For questionable water sources, combine methods:

1. **Pre-filter** (cloth, coffee filter)
2. **Filter** (commercial or DIY)
3. **Disinfect** (boil, chemical, or UV)
4. **Store safely** (covered, clean container)
